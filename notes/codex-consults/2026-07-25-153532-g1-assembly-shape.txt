"""Executable checker for the assembly_shape G1 spec.

Runs the shipped classifier (builders/special.py logic, reproduced via
the same clique-sum recursion) over all 11 type-1 members and asserts:
  (T) tree-ness: sum of dual degrees == 2(n-1)  [dual graph of a
      clique-sum of spheres along separating triangles is a tree;
      catches spurious |V_i ∩ V_j|=3 adjacencies]
  (S) shape matches the independently-known structure of each member:
      helix family -> path; n-around-an-axis-edge -> around an edge
      with the common pair = the axis pair; stars/branched forms ->
      branched.
  (O) check order: every 'branched' verdict would NOT also satisfy the
      around-an-edge test (|common| != 2), i.e. the precedence never
      hides an ambiguity on real members.
"""
import json, sys
from itertools import combinations

def factors_with_verts(faces):
    def rec(fs):
        fset = {frozenset(f) for f in fs}
        adj = {}
        for a, b, c in fs:
            adj.setdefault(a, set()).update((b, c))
            adj.setdefault(b, set()).update((a, c))
            adj.setdefault(c, set()).update((a, b))
        seps = [t for t in {frozenset(t) for v in adj
                for t in combinations(sorted(adj[v] | {v}), 3)
                if all(y in adj[x] for x, y in combinations(t, 2))}
                if t not in fset]
        if not seps:
            return [fs]
        T = sorted(min(seps, key=lambda t: tuple(sorted(t))))
        cut = {frozenset(e) for e in combinations(T, 2)}
        e2f = {}
        for i, f in enumerate(fs):
            for e in zip(f, f[1:] + f[:1]):
                e2f.setdefault(frozenset(e), []).append(i)
        fadj = {i: set() for i in range(len(fs))}
        for e, ff in e2f.items():
            if e in cut: continue
            for i in ff: fadj[i].update(j for j in ff if j != i)
        comps, seen = [], set()
        for i in range(len(fs)):
            if i in seen: continue
            comp, st = set(), [i]
            while st:
                k = st.pop()
                if k not in comp: comp.add(k); st.extend(fadj[k] - comp)
            seen |= comp; comps.append(comp)
        out = []
        for comp in comps:
            out.extend(rec([fs[i] for i in sorted(comp)] + [tuple(T)]))
        return out
    return [frozenset(v for f in fac for v in f) for fac in rec(faces)]

def shape(fv):
    if len(fv) < 3:
        return None
    dd = [sum(1 for j, w in enumerate(fv) if j != i and len(v & w) == 3)
          for i, v in enumerate(fv)]
    if max(dd) > 2:
        return 'branched'
    if len(frozenset.intersection(*fv)) == 2:
        return 'around an edge'
    return 'path'

# expected shapes from independent structural knowledge:
# helix family (CLERS (CA)^k pattern) = path; the 3- and 4-around-an-
# axis-edge forms; the partial/full stars and the star+leaf forms.
EXPECT = {
    'v5CCACAE':                    (2, None),
    'v6CCACACAE':                  (3, 'around an edge'),
    'v7CCACACACAE':                (4, 'path'),
    'v7CCACACCABE':                (4, 'around an edge'),
    'v7CCACCABCAE':                (4, 'branched'),
    'v8CCACACACACAE':              (5, 'path'),
    'v8CCACACCACABE':              (5, 'branched'),
    'v8CCACCCABCABE':              (5, 'branched'),
    'v9CCACACACACACAE':            (6, 'path'),
    'v10CCACACACACACACAE':         (7, 'path'),
    'v14CCACACACACACACACACACACAE': (11, 'path'),
}

fail = 0
for nid, (n_exp, sh_exp) in EXPECT.items():
    rec = json.load(open(f'site/personal/nets/{nid}/net.json'))
    faces = [tuple(int(x) for x in f.split(','))
             for f in rec['netcode'].split(';')]
    fv = factors_with_verts(faces)
    n = len(fv)
    dd = [sum(1 for j, w in enumerate(fv) if j != i and len(v & w) == 3)
          for i, v in enumerate(fv)]
    tree_ok = (sum(dd) == 2 * (n - 1))
    sh = shape(fv)
    order_ok = True
    if sh == 'branched':
        order_ok = (len(frozenset.intersection(*fv)) != 2)
    axis_ok = True
    if sh == 'around an edge':
        common = frozenset.intersection(*fv)
        # the common pair must be an edge: both vertices in every factor,
        # and (in a tet list) adjacent -- every factor is a tet containing
        # both, so they are adjacent in it by definition of a tet.
        axis_ok = (len(common) == 2)
    ok = (n == n_exp and sh == sh_exp and tree_ok and order_ok and axis_ok)
    fail += 0 if ok else 1
    print(f"{'OK ' if ok else 'FAIL'} {nid:30} n={n:2} (exp {n_exp:2}) "
          f"shape={str(sh):15} (exp {str(sh_exp):15}) "
          f"tree={'Y' if tree_ok else 'N'} order={'Y' if order_ok else 'N'}")
print(f"\n{len(EXPECT) - fail}/{len(EXPECT)} members pass all assertions")
sys.exit(1 if fail else 0)
